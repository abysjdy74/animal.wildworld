    const grid =
        document.getElementById("animalGrid");

    grid.innerHTML = "";

    const filtered = animals.filter(animal => {

        const matchesCategory =
            currentCategory === "All" ||
            animal.category === currentCategory;

        const matchesSearch =
            animal.name.toLowerCase().includes(searchText) ||
            animal.habitat.toLowerCase().includes(searchText) ||
            animal.diet.toLowerCase().includes(searchText);

        return matchesCategory && matchesSearch;

    });

    document.getElementById("count").innerText =
        animals.length;

    if(filtered.length === 0) {

        grid.innerHTML =
            "<h3>No animals found! Try another search 🐾</h3>";

        return;

    }

    filtered.forEach(animal => {

        grid.innerHTML += createCard(animal);

    });

    renderFavorites();

}

// CREATE ANIMAL CARD

function createCard(animal) {

    const isSaved = favorites.includes(animal.name);

    return `

    <div class="card">

        <div class="animal-emoji">

            ${animal.emoji}

            <button class="heart"
            onclick="toggleFavorite('${animal.name}')">

            ${isSaved ? "❤️" : "🤍"}

            </button>

        </div>

        <div class="card-content">

            <span class="tag">
                ${animal.category}
            </span>

            <h3>${animal.name}</h3>

            <p>${animal.fact}</p>

            <p>
                📍 ${animal.habitat}
            </p>

            <button class="details"
            onclick="showDetails('${animal.name}')">

                Discover More ➜

            </button>

        </div>

    </div>

    `;

}

// FAVORITES

function toggleFavorite(name) {

    if(favorites.includes(name)) {

        favorites = favorites.filter(
            animal => animal !== name
        );

    } else {

        favorites.push(name);

    }

    localStorage.setItem(
        "wildFavorites",
        JSON.stringify(favorites)
    );

    renderAnimals();

}

function renderFavorites() {

    const grid =
        document.getElementById("favoriteGrid");

    grid.innerHTML = "";

    const savedAnimals = animals.filter(
        animal => favorites.includes(animal.name)
    );

    if(savedAnimals.length === 0) {

        grid.innerHTML =
        "<p>No favorites yet! Click a heart to save an animal ❤️</p>";

        return;

    }

    savedAnimals.forEach(animal => {

        grid.innerHTML += createCard(animal);

    });

}

let questionIndex = 0;

function showQuestion() {

    const question =
        questions[questionIndex];

    document.getElementById("question")
    .innerText = question.question;

    const answers =
        document.getElementById("answers");

    answers.innerHTML = "";

    document.getElementById("result")
    .innerText = "";

    question.answers.forEach((answer, index) => {

        let button =
            document.createElement("button");

        button.innerText = answer;

        button.onclick = function() {

            checkAnswer(index);

        };

        answers.appendChild(button);

    });

}

function checkAnswer(index) {

    const question =
        questions[questionIndex];

    const result =
        document.getElementById("result");

    if(index === question.correct) {

        result.innerText =
            "🎉 Correct! " + question.explanation;

    } else {

        result.innerText =
            "❌ Not quite! " + question.explanation;

    }

    document.querySelectorAll("#answers button")
    .forEach(button => button.disabled = true);

}

function nextQuestion() {

    questionIndex++;

    if(questionIndex >= questions.length) {

        questionIndex = 0;

    }

    showQuestion();

}
